#
# IMPORTANT: conversion environment must use the project Python version (3.12.x).

# which causes pip to compile onnx from source and fail on macOS.

onnx==1.16.2; python_version >= "3.12" and python_version < "3.13"
onnxruntime==1.20.1; python_version >= "3.12" and python_version < "3.13"
